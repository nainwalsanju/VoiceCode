```python
def get_voice_state_for_engine(
    self,
    voice_source: str,
    engine: str,
) -> tuple[any, str]:
    """
    Get voice state optimized for a specific TTS engine.
    
    Args:
        voice_source: Preset voice name or profile ID
        engine: TTS engine name (qwen3, neutts-air, etc.)
    
    Returns:
        Tuple of (voice_state, voice_type)
    """
    # For Pocket TTS, use existing method
    if engine == "pocket":
        return self.get_voice_state(voice_source)
    
    # For engines that support voice cloning, generate appropriate state
    engine_states = self._engine_voice_states.get(engine, {})
    cache_key = f"{engine}:{voice_source}"
    
    if cache_key in engine_states:
        return engine_states[cache_key], "cloned"
    
    # Load the audio and create engine-specific voice state
    voice_state = self._create_engine_voice_state(voice_source, engine)
    engine_states[cache_key] = voice_state
    self._engine_voice_states[engine] = engine_states
    
    return voice_state, "cloned"
```

```python
def _create_engine_voice_state(self, voice_source: str, engine: str) -> any:
```

```python
def _get_audio_data(self, voice_source: str) -> bytes:
    """Get audio data from voice source."""
    if voice_source in AVAILABLE_PRESET_VOICES:
        return self._get_preset_audio(voice_source)
    
    ref_path = VOICE_STATES_DIR / f"{voice_source}.json"
    if ref_path.exists():
        ref_data = json.loads(ref_path.read_text())
        audio_path = ref_data.get("audio_source")
        if audio_path and Path(audio_path).exists():
            return Path(audio_path).read_bytes()
    
    raise ValueError(f"Cannot find audio for voice source: {voice_source}")

def _create_qwen3_voice_state(self, audio_data: bytes) -> any:
    """Create voice state for Qwen3 TTS."""
    try:
        from qwen_tts import VoiceEncoder
        encoder = VoiceEncoder()
        return encoder.encode_reference(audio_data)
    except ImportError:
        logger.warning("qwen_tts not available for voice cloning")
        return None

def _create_neutts_voice_state(self, audio_data: bytes, model_type: str) -> any:
    """Create voice state for NeuTTS."""
    try:
        from neursson import VoiceEncoder
        encoder = VoiceEncoder(model_type=model_type)
        return encoder.encode_speaker(audio_data)
    except ImportError:
        logger.warning(f"neursson {model_type} not available for voice cloning")
        return None

def _create_kokoro_voice_state(self, audio_data: bytes) -> any:
    """Create voice state for Kokoro."""
    try:
        from kokoro import VoiceEncoder
        encoder = VoiceEncoder()
        return encoder.encode_voice(audio_data)
    except ImportError:
        logger.warning("kokoro not available for voice cloning")
        return None
```

```python
async def generate_audio_async(
    self,
    text: str,
    voice: str = DEFAULT_VOICE,
    rate: str = DEFAULT_RATE,
    pitch: str = DEFAULT_PITCH,
    voice_profile_id: Optional[str] = None,
) -> tuple[bytes, int]:
    if not text or not text.strip():
        raise ValueError("Text cannot be empty")

    self._load_model(voice)
    logger.info("generating_audio", text_length=len(text), voice=voice, voice_profile_id=voice_profile_id)

    audio_bytes = b""

    try:
        # Use cloned voice if profile provided
        if voice_profile_id and self._should_use_cloned_voice(voice_profile_id, voice):
            voice_clone_service = get_voice_cloning_service()
            voice_state, _ = voice_clone_service.get_voice_state_for_engine(voice_profile_id, voice.split("-")[0])
            
            if voice_state:
                audio_bytes = self._generate_with_cloned_voice(text, voice, voice_state)
            else:
                audio_bytes = self._generate_without_clone(text, voice)
        else:
            audio_bytes = self._generate_without_clone(text, voice)
            
    except Exception as e:
        logger.error("tts_generation_error", error=str(e))
        audio_bytes = self._generate_silence(len(text))

    duration_ms = int(len(audio_bytes) / 32) if audio_bytes else 0
    return audio_bytes, duration_ms
```

```python
def _generate_with_cloned_voice(self, text: str, voice: str, voice_state: any) -> bytes:
    """Generate audio using cloned voice state."""
    try:
        if voice.startswith("qwen3") and self._qwen3_model:
            audio = self._qwen3_model.generate_with_voice(text, voice_state)
            return self._audio_to_bytes(audio)
        elif voice.startswith("neutts-air") and self._neutts_air_model:
            audio = self._neutts_air_model.speak_with_voice(text, voice_state)
            return self._audio_to_bytes(audio)
        elif voice.startswith("neutts-nano") and self._neutts_nano_model:
            audio = self._neutts_nano_model.speak_with_voice(text, voice_state)
            return self._audio_to_bytes(audio)
        elif voice.startswith("pocket") and self._pocket_model:
            audio = self._pocket_model.generate(text, voice_state=voice_state)
            return self._audio_to_bytes(audio)
        elif voice.startswith("kokoro") and self._kokoro_model:
            audio = self._kokoro_model.speak_with_voice(text, voice_state)
            return self._audio_to_bytes(audio)
        else:
            return self._generate_silence(len(text))
    except Exception as e:
        logger.error("cloned_voice_generation_error", error=str(e))
        return self._generate_silence(len(text))

def _generate_without_clone(self, text: str, voice: str) -> bytes:
    """Generate audio without cloned voice."""
    if self._qwen3_model:
        audio = self._qwen3_model.generate(text)
        return self._audio_to_bytes(audio)
    elif self._pocket_model:
        audio = self._pocket_model.generate(text)
        return self._audio_to_bytes(audio)
    elif self._kokoro_model:
        audio = self._kokoro_model.generate(text)
        return self._audio_to_bytes(audio)
    else:
        return self._generate_silence(len(text))
```

```python
        logger.error("tts_stream_error", error=str(e))
```

```python
@router.post("/synthesize")
async def synthesize_speech(
    text: str,
    voice: str = "qwen3-tts-0.6b",
    voice_profile_id: Optional[str] = None,
):
    tts_service = get_tts_service()
    
    audio_bytes, duration_ms = await tts_service.generate_audio_async(
        text,
        voice=voice,
        voice_profile_id=voice_profile_id,
    )
    
    return Response(
        content=audio_bytes,
        media_type="audio/wav",
        headers={
            "X-Audio-Duration-Ms": str(duration_ms),
            "X-Voice-Profile-Id": voice_profile_id or "",
        }
    )
```

2. Update streaming endpoint similarly:
```python
@router.websocket("/stream")
async def stream_speech(websocket: WebSocket):
    await websocket.accept()
    
    try:
        data = await websocket.receive_json()
        text = data.get("text", "")
        voice = data.get("voice", "qwen3-tts-0.6b")
        voice_profile_id = data.get("voice_profile_id")
        
        tts_service = get_tts_service()
        
        async for chunk in tts_service.stream_audio_async(
            text,
            voice=voice,
            voice_profile_id=voice_profile_id,
        ):
            await websocket.send_bytes(chunk)
            
    except Exception as e:
        logger.error("stream_error", error=str(e))
    finally:
        await websocket.close()
```
