```typescript
useEffect(() => {
  let cleanup: (() => void) | null = null;

  const setupHotkey = async () => {
    const settings = await getSettings();
    const hotkey = settings.hotkey || 'Ctrl+Shift+V';

    const handleHotkeyActivation = () => {
      setHotkeyActivated(true);
      setTimeout(() => setHotkeyActivated(false), 500);

      const currentState = useSessionStore.getState().currentState;

      if (currentState === 'IDLE') {
        startInteraction();
      } else {
        stopInteraction();
      }
    };

    await registerGlobalHotkey(hotkey, handleHotkeyActivation);

    cleanup = () => {
      unregisterGlobalHotkey(hotkey);
    };
  };

  setupHotkey().catch(console.error);

  return () => {
    cleanup?.();
  };
}, []);
```

2. In handleSave, add special handling for hotkey:
```typescript
const handleSave = async (key: string, value: unknown) => {
  if (!settings) return;

  try {
    if (key === 'hotkey' && settings.hotkey) {
      await unregisterGlobalHotkey(settings.hotkey);
    }

    const updated = await updateSettings({ [key]: value });
    setSettings(updated);
  } catch (err) {
    setError(err instanceof Error ? err.message : 'Datalink failure: Synchronization failed');
  }
};
```

3. Add hotkey format normalizer:
```typescript
const normalizeHotkey = (input: string): string => {
  return input
    .toUpperCase()
    .replace(/CONTROL/gI, 'CTRL')
    .replace(/COMMAND|CMD|META/g, 'SUPER')
    .replace(/\s+/g, '');
};
```

4. Use in input onChange:
```typescript
onChange={(e) => handleSave('hotkey', normalizeHotkey(e.target.value))}
```
