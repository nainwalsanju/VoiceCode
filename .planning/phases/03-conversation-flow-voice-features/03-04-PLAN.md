```typescript
const getAudioDuration = (file: File): Promise<number> => {
  return new Promise((resolve, reject) => {
    const audio = new Audio();
    audio.preload = 'metadata';
    
    audio.onloadedmetadata = () => {
      URL.revokeObjectURL(audio.src);
      resolve(audio.duration);
    };
    
    audio.onerror = () => {
      URL.revokeObjectURL(audio.src);
      reject(new Error('Failed to load audio'));
    };
    
    audio.src = URL.createObjectURL(file);
  });
};
```

2. Create a validation function:
```typescript
const validateAudioDuration = async (file: File): Promise<{ valid: boolean; duration: number; error?: string }> => {
  try {
    const duration = await getAudioDuration(file);
    
    if (duration < 2) {
      return { valid: false, duration, error: 'Sample too short: Minimum 2 seconds required.' };
    }
    
    if (duration > 15) {
      return { valid: false, duration, error: 'Sample too long: Maximum 15 seconds allowed.' };
    }
    
    return { valid: true, duration };
  } catch {
    return { valid: false, duration: 0, error: 'Unable to read audio file.' };
  }
};
```

```typescript
    setAudioFile(null);
    setAudioUrl(null);
    onSuccess?.();

};
```
